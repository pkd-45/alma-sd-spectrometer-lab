cmake_minimum_required(VERSION 3.16)
project(alma_fft_reference LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(alma_fft_reference src/fft_reference.cpp)
target_compile_options(alma_fft_reference PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
add_test(NAME tone_recovery COMMAND alma_fft_reference --self-test)
